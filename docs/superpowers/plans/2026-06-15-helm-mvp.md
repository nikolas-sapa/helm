# Helm MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the thinnest end-to-end slice of "Vercel for internal agents": an employee runs `helm deploy`, gets a hosted HTTP-triggered Claude agent plus an auto-provisioned Convex DB, and IT sees every run, scopes tools/domains, and tracks token spend per agent.

**Architecture:** npm-workspaces monorepo. Pure deterministic logic (cost, policy, hashing, domain gating) lives in `packages/core` and `packages/agent` and is fully unit-tested. Infra glue (Vercel Sandbox runner, Convex provisioner) is isolated behind narrow interfaces so it can be stubbed in tests and validated by a Step-0 spike. A Next.js app (`apps/dashboard`) is both the IT control plane UI and the HTTP ingress/runner. Control-plane state lives in Convex (dogfooding).

**Tech Stack:** TypeScript, npm workspaces, Vitest, Next.js (App Router), Convex `1.41.0`, `@anthropic-ai/claude-agent-sdk@0.3.177`, `@vercel/sandbox@2.2.1`, `@anthropic-ai/sdk@0.104.1`, Commander (CLI).

**Spec:** `docs/superpowers/specs/2026-06-15-helm-mvp-design.md`

import type { Usage } from "./types.js";

// USD per 1M tokens. Update from Anthropic pricing; keyed by exact model id.
export const PRICES: Record<string, { in: number; out: number }> = {
  "claude-opus-4-8": { in: 5, out: 25 },
  "claude-sonnet-4-6": { in: 3, out: 15 },
  "claude-haiku-4-5-20251001": { in: 1, out: 5 },
};

export function computeCost(u: Usage): number {
  const p = PRICES[u.model];
  if (!p) throw new Error(`unknown model: ${u.model}`);
  return (u.tokensIn / 1_000_000) * p.in + (u.tokensOut / 1_000_000) * p.out;
}
```

import type { Policy } from "./types.js";

export function evaluateTool(policy: Policy, tool: string): boolean {
  return policy.allowedTools.includes(tool);
}

export function evaluateDomain(policy: Policy, url: string): boolean {
  let host: string;
  try { host = new URL(url).hostname.toLowerCase(); } catch { return false; }
  return policy.allowedDomains.some((d) => host === d.toLowerCase());
}

export interface BudgetState { runTokens: number; monthTokens: number; }
export type BudgetVerdict = { ok: true } | { ok: false; reason: "per_run_ceiling" | "monthly_cap" };

export function evaluateBudget(policy: Policy, s: BudgetState): BudgetVerdict {
  if (s.runTokens > policy.perRunTokenCeiling) return { ok: false, reason: "per_run_ceiling" };
  if (s.monthTokens >= policy.monthlyTokenCap) return { ok: false, reason: "monthly_cap" };
  return { ok: true };
}
```
Note: exact-host match only — no suffix matching — which is why the lookalike-subdomain test passes.

import { createHash } from "node:crypto";

export interface BundleFile { path: string; content: string; }

export function hashBundle(files: BundleFile[]): string {
  const h = createHash("sha256");
  for (const f of [...files].sort((a, b) => a.path.localeCompare(b.path))) {
    h.update(f.path); h.update("\0"); h.update(f.content); h.update("\0");
  }
  return h.digest("hex");
}
```

import { evaluateDomain, type Policy } from "@helm/core";

export type FetchFn = (url: string, init?: RequestInit) => Promise<Response>;

export function makeFetchTool(policy: Policy, under: FetchFn = fetch): FetchFn {
  return async (url, init) => {
    if (!evaluateDomain(policy, url)) throw new Error(`domain not allowed: ${url}`);
    return under(url, init);
  };
}

export interface ProvisionApi {
  createProject(agentId: string): Promise<{ projectId: string; url: string; adminKey: string }>;
}
export interface Shared { url: string; adminKey: string; }
export interface Provisioned { projectId?: string; url: string; adminKey: string; namespace?: string; }

export async function provisionWith(api: ProvisionApi, agentId: string, shared?: Shared): Promise<Provisioned> {
  try {
    return await api.createProject(agentId);
  } catch (e) {
    if (!shared) throw e;
    return { url: shared.url, adminKey: shared.adminKey, namespace: agentId.replace(/[^a-z0-9]/gi, "_").toLowerCase() };
  }
}

import { computeCost, type RunResult, type ToolCall, type Usage } from "@helm/core";

const SECRET_RE = /sk-ant-[A-Za-z0-9_-]+/g;

export function redactInput(input: string): string {
  return input.replace(SECRET_RE, "[redacted]").slice(0, 200);
}

export interface SandboxOutcome { output: unknown; usage: Usage; toolCalls: ToolCall[]; durationMs: number; error?: string; }

export function assembleResult(o: SandboxOutcome): RunResult {
  const status = o.error ? "failed" : o.toolCalls.some((t) => !t.allowed) ? "blocked" : "ok";
  return { status, output: o.output, usage: o.usage, costUsd: computeCost(o.usage), toolCalls: o.toolCalls, durationMs: o.durationMs, error: o.error };
}
```

import { createHash, timingSafeEqual } from "node:crypto";

export function hashKey(key: string): string {
  return createHash("sha256").update(key).digest("hex");
}
export function verifyKey(key: string, hash: string): boolean {
  const a = Buffer.from(hashKey(key), "hex");
  const b = Buffer.from(hash, "hex");
  return a.length === b.length && timingSafeEqual(a, b);
}
